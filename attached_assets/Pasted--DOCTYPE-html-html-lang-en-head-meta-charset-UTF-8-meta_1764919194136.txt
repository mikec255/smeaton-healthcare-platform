        // Matrix data: { memberId: { taskId: { status, date } } }
        const matrixData = {

        function renderMatrix() {
            const grid = document.getElementById('matrixGrid');
            const columnCount = tasks.length;
            grid.style.setProperty('--columns', columnCount);
            
            let html = '';
            
            // Header row
            html += '<div class="matrix-cell header-cell corner-cell name-cell">Team Member</div>';
            tasks.forEach(task => {
                html += '<div class="matrix-cell header-cell">' + task.name + '</div>';
            });
            
            // Data rows
            teamMembers.forEach(member => {
                // Name cell
                html += '<div class="matrix-cell name-cell">' +
                    '<div class="name-content">' +
                    '<div class="avatar">' + member.initials + '</div>' +
                    '<div class="name-info">' +
                    '<div class="name-text">' + member.name + '</div>' +
                    '<div class="role-text">' + member.role + '</div>' +
                    '</div></div></div>';
                
                // Task cells
                tasks.forEach(task => {
                    const cellData = matrixData[member.id][task.id] || { status: 'none', date: null };
                    const tooltip = task.name + ' - ' + cellData.status.replace('-', ' ');
                    
                    html += '<div class="matrix-cell">' +
                        '<div class="status-content">' +
                        '<div class="status-box status-' + cellData.status + '" ' +
                        'title="' + tooltip + '" ' +
                        'onclick="toggleStatus(' + member.id + ', \'' + task.id + '\')"></div>';
                    
                    if (cellData.date) {
                        html += '<span class="date-text">' + cellData.date + '</span>';
                    }
                    
                    html += '</div></div>';
                });
            });
            
            grid.innerHTML = html;
        }
        
        function toggleStatus(memberId, taskId) {
            const statuses = ['none', 'in-progress', 'completed', 'overdue', 'blocked'];
            const currentStatus = matrixData[memberId][taskId].status;
            const currentIndex = statuses.indexOf(currentStatus);
            const nextIndex = (currentIndex + 1) % statuses.length;
            
            matrixData[memberId][taskId].status = statuses[nextIndex];
            
            if (statuses[nextIndex] === 'completed') {
                const today = new Date();
                matrixData[memberId][taskId].date = 
                    String(today.getDate()).padStart(2, '0') + '/' + 
                    String(today.getMonth() + 1).padStart(2, '0');
            } else if (statuses[nextIndex] === 'none') {
                matrixData[memberId][taskId].date = null;
            }
            
            renderMatrix();
        }
        
        // Initialize
        document.addEventListener('DOMContentLoaded', renderMatrix);
